#include <sys/types.h>
#include <cstdint>
#include <cstdio>
#include <limits>
#include <type_traits>

int main() {
    std::printf("off_t_bits=%zu ssize_t_bits=%zu size_t_bits=%zu long_bits=%zu\n",
                sizeof(off_t) * 8u, sizeof(ssize_t) * 8u, sizeof(size_t) * 8u, sizeof(long) * 8u);
    std::printf("off_t_max=%lld ssize_t_max=%lld\n",
                static_cast<long long>(std::numeric_limits<off_t>::max()),
                static_cast<long long>(std::numeric_limits<ssize_t>::max()));
    std::printf("off_t_is_long=%d ssize_t_is_long=%d\n",
                static_cast<int>(std::is_same_v<off_t, long>),
                static_cast<int>(std::is_same_v<ssize_t, long>));
#ifdef _FILE_OFFSET_BITS
    std::printf("file_offset_bits_macro=%d\n", _FILE_OFFSET_BITS);
#else
    std::printf("file_offset_bits_macro=unset\n");
#endif
    return 0;
}
